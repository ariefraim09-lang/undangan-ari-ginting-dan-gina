rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }
    match /Ucapan/{id} {
      allow read: if resource.data.status == "approved" || isAdmin();
      allow create: if
        request.resource.data.keys().hasAll(["Nama","Pesan","Likes","status","createdAt"]) &&
        request.resource.data.keys().hasOnly(["Nama","Pesan","Likes","status","createdAt"]) &&
        request.resource.data.Nama is string && request.resource.data.Nama.size() > 0 && request.resource.data.Nama.size() <= 80 &&
        request.resource.data.Pesan is string && request.resource.data.Pesan.size() > 0 && request.resource.data.Pesan.size() <= 1000 &&
        request.resource.data.Likes is int && request.resource.data.Likes == 0 &&
        request.resource.data.status == "pending" && request.resource.data.createdAt is timestamp;
      allow update: if isAdmin() || (
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(["Likes"]) &&
        request.resource.data.Likes is int && request.resource.data.Likes == resource.data.Likes + 1
      );
      allow delete: if isAdmin();
    }
    match /admins/{uid} { allow read, write: if isAdmin(); }
    match /{document=**} { allow read, write: if false; }
  }
}
